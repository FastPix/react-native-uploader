    @objc private func showQualityOptions() {
       
        let levels = playerViewController.getResolutionLevels()
        let current = playerViewController.getCurrentResolutionLevel()
       
        guard !levels.isEmpty else {
            return
        }
       
        let alert = UIAlertController(
            title: "Video Quality",
            message: nil,
            preferredStyle: .actionSheet
        )
       
        for level in levels {
           
            let isSelected = level.id == current?.id
            let title = isSelected ? "✓  \(level.label)" : "    \(level.label)"
           
            alert.addAction(UIAlertAction(title: title, style: .default) { _ in
               
                if level.isAuto {
                    self.playerViewController.resetToAuto()
                } else {
                    self.playerViewController.setResolutionLevel(level)
                }
            })
        }
       
        alert.addAction(UIAlertAction(title: "Cancel", style: .cancel))
       
        present(alert, animated: true)
    }

    @objc private func showAudioTracks() {
       
        let audioTracks    = playerViewController.getAudioTracks()
        let selectedAudio  = playerViewController.getCurrentAudioTrack()
        let subtitleTracks = playerViewController.getSubtitleTracks()
        let selectedSub    = playerViewController.getCurrentSubtitleTrack()
       
        let alert = UIAlertController(
            title: "Settings",
            message: nil,
            preferredStyle: .actionSheet
        )
       
        // MARK: - Audio section header
        alert.addAction(UIAlertAction(
            title: "Audio Tracks",
            style: .default,
            handler: nil
        ))
       
        for track in audioTracks {
            let isSelected = track.id == selectedAudio?.id
            let title = isSelected ? "✓  \(track.label)" : "    \(track.label)"
            alert.addAction(UIAlertAction(title: title, style: .default) { _ in
                self.playerViewController.setAudioTrack(trackId: track.id)
            })
        }
       
        // MARK: - Subtitle section header
        alert.addAction(UIAlertAction(
            title: "Subtitles",
            style: .default,
            handler: nil
        ))
       
        // Off option
        alert.addAction(UIAlertAction(
            title: selectedSub == nil ? "✓  Off" : "    Off",
            style: .default
        ) { _ in
            self.playerViewController.disableSubtitles()
        })
       
        for track in subtitleTracks {
            let isSelected = track.id == selectedSub?.id
            let title = isSelected ? "✓  \(track.label)" : "    \(track.label)"
            alert.addAction(UIAlertAction(title: title, style: .default) { _ in
                try? self.playerViewController.setSubtitleTrack(trackId: track.id)
            })
        }
       
        alert.addAction(UIAlertAction(title: "Cancel", style: .cancel))
        present(alert, animated: true)
    }

    enum SeekDirection {
        case forward
        case backward
    }
   
    private func showSeekOverlay(for direction: SeekDirection) {
        let overlay = (direction == .forward) ? rightSeekOverlay : leftSeekOverlay
        UIView.animate(withDuration: 0.2, animations: {
            overlay?.alpha = 1
        }) { _ in
            UIView.animate(withDuration: 0.2, delay: 0.2, options: [], animations: {
                overlay?.alpha = 0
            })
        }
    }
   
    private func createSeekOverlay(
        iconName: String,
        text: String
    ) -> UIView {
       
        let container = UIView()
        container.translatesAutoresizingMaskIntoConstraints = false
        container.layer.cornerRadius = 12
        container.alpha = 0
       
        let stack = UIStackView()
        stack.axis = .vertical
        stack.alignment = .center
        stack.spacing = 6
        stack.translatesAutoresizingMaskIntoConstraints = false
       
        let icon = UIImageView()
        icon.image = UIImage(systemName: iconName)
        icon.tintColor = .white
        icon.contentMode = .scaleAspectFit
       
        let label = UILabel()
        label.text = text
        label.textColor = .white
        label.font = UIFont.boldSystemFont(ofSize: 14)
       
        stack.addArrangedSubview(icon)
        stack.addArrangedSubview(label)
        container.addSubview(stack)
       
        NSLayoutConstraint.activate([
            stack.centerXAnchor.constraint(equalTo: container.centerXAnchor),
            stack.centerYAnchor.constraint(equalTo: container.centerYAnchor),
            container.widthAnchor.constraint(equalToConstant: 60),
            container.heightAnchor.constraint(equalToConstant: 50)
        ])
        return container
    }

    private func showLoader() {

    @objc private func handleSeekGesture(_ notification: Notification) {
        guard let direction = notification.userInfo?["direction"] as? String else { return }
       
        if direction == "forward" {
            showSeekOverlay(for: .forward)
        } else {
            showSeekOverlay(for: .backward)
        }
    }
   
    override func viewDidAppear(_ animated: Bool) {
        super.viewDidAppear(animated)
       
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(handleSeekGesture(_:)),
            name: Notification.Name("fastpixSeekGesture"),
            object: nil
        )
       
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) {
            self.playerViewController.setupPiP(parent: self)
        }
       
        guard let popGesture = navigationController?.interactivePopGestureRecognizer,
              let sliderPan = seekBar.panGestureRecognizer else {
            return
        }
       
        popGesture.require(toFail: sliderPan)
       
        guard let popGesture = navigationController?.interactivePopGestureRecognizer,
              let sliderPan = seekBar.panGestureRecognizer else { return }
       
        popGesture.require(toFail: sliderPan)

    private func autoHideControls(after delay: TimeInterval = 3.0) {
        controlsHideWorkItem?.cancel()
       
        guard !isUserSeeking else { return } // ✅ already there (good)
       
        let workItem = DispatchWorkItem { [weak self] in
            guard let self else { return }
            guard !self.isUserSeeking else { return }

        DispatchQueue.main.asyncAfter(deadline: .now() + delay, execute: workItem)
    }

    @objc private func playlistStateChanged(_ notification: Notification) {
        self.showLoader()
        DispatchQueue.main.async {
            self.resetSubtitleState()
            self.seekBar.resetBuffer()
            // 🔹 Clear spritesheet
            self.playerViewController.clearSpritesheet()

            self.updateCurrentTitle()
            self.updateButtonVisibility()
            self.updatePlayPauseButton(for: self.playerViewController.player?.timeControlStatus ?? .paused)
           
            // 2️⃣ ALWAYS reset skip state
            self.playerViewController.skipManager?.clearSegments()
            self.hideSkipButton()
           
            // 3️⃣ Extract playlist item SAFELY
            guard let item = notification.object as? FastPixPlaylistItem else {
                return
            }
           
            if !item.skipSegments.isEmpty {
                self.playerViewController.skipManager?.setSkipSegments(item.skipSegments)
            }
           
            self.playerViewController.qualityManager?.loadQualityLevels()
           
            let previewConfig = FastPixSeekPreviewConfig()
            self.playerViewController.loadSpritesheet(
                url: nil,
                previewEnable: true,
                config: previewConfig
            )
            self.playerViewController.setFallbackMode(.timestamp)
           
            // Preload upcoming videos after playlist item changes
            self.preloadNextVideos()
            self.precacheUpcomingVideos()
        }
    }

    @objc private func nextTapped(_ sender: UIButton) {
        sender.animateTapFeedback()
        userDidInteract()
       
        // Check if the next item is already preloaded — consuming it detaches
        // it from the shadow player so AVFoundation's URL cache can reuse the
        // buffered data when the SDK loads the same URL via next().
        let nextIndex = playerViewController.currentPlaylistIndex + 1
        if nextIndex < playlist.count {
            let nextId = playlist[nextIndex].playbackId
            // Consume (detach from shadow player) so cached data is available
            if preloadManager.consumePreloadedItem(for: nextId) != nil {
                print("Preloaded item consumed — AVFoundation cache warm for: \(nextId)")
            }
        }
        _ = playerViewController.next()
       
        resetSubtitleState()
        seekBar.resetBuffer()
        self.playerViewController.clearSpritesheet()
        self.seekBar.previewView.isHidden = true
        self.playerViewController.setupSeekManager(delegate: self)
        updateCurrentTitle()
        updateButtonVisibility()
        self.updatePlayPauseButton(for: playerViewController.player?.timeControlStatus ?? .paused)
        let previewConfig = FastPixSeekPreviewConfig()
        playerViewController.loadSpritesheet(url: nil, previewEnable: true, config: previewConfig)
        playerViewController.setFallbackMode(.timestamp)
        preloadNextVideos()
        precacheUpcomingVideos()
    }

// MARK: - Preload & Precache Helpers

extension VideoPlayerViewController {
   
    /// Preloads the next 2 videos after the currently playing item.
    /// Call this whenever the current playlist position changes.
    private func preloadNextVideos() {
        let currentIndex = playerViewController.currentPlaylistIndex
        let currentId = currentIndex < playlist.count ? playlist[currentIndex].playbackId : ""
       
        let upcoming = playlist
            .enumerated()
            .filter { $0.offset > currentIndex }
            .prefix(2)
            .map { $0.element }
            .filter {
                // Skip if already loading, ready, or currently playing
                let status = preloadManager.preloadStatus(forVideo: $0.playbackId)
                switch status {
                case .idle: return $0.playbackId != currentId
                default:    return false
                }
            }
       
        let itemsToPreload: [(id: String, item: AVPlayerItem)] = upcoming.compactMap { playlistItem -> (id: String, item: AVPlayerItem)? in
            guard let url = buildPlaybackURL(for: playlistItem) else { return nil }
            let playerItem = AVPlayerItem(url: url)
            return (id: playlistItem.playbackId, item: playerItem)
        }
       
        guard !itemsToPreload.isEmpty else { return }
        preloadManager.preload(items: itemsToPreload)
        print("Queued preload for \(itemsToPreload.map(\.id))")
    }
   
    /// Precaches HLS segments to disk for the current + next video.
    /// Skips DRM items (token-protected) since their segments are encrypted.
    private func precacheUpcomingVideos() {
        let currentIndex = playerViewController.currentPlaylistIndex
        let upcoming = playlist
            .enumerated()
            .filter { $0.offset >= currentIndex }
            .prefix(2)
            .map { $0.element }
       
        for item in upcoming {
            // Skip DRM-protected content — encrypted segments cannot be cached
            guard item.drmToken.isEmpty else {
                print("Skipping precache for DRM item: \(item.playbackId)")
                continue
            }
           
            let host = item.customDomain.isEmpty == false
            ? item.customDomain
            : "stream.fastpix.com"
           
            var urlString = "https://\(host)/\(item.playbackId).m3u8"
            if !item.token.isEmpty {
                urlString += "?token=\(item.token)"
            }
           
            guard let url = URL(string: urlString) else { continue }
            precacheManager.startPrecaching(url: url)
            print("Precaching: \(item.playbackId)")
        }
    }
   
    /// Builds the correct PlaybackOptions for a given playlist item.
    private func buildPlaybackOptions(for item: FastPixPlaylistItem) -> PlaybackOptions {
        let hasDomain = !(item.customDomain.isEmpty ?? true)
        let hasToken  = !item.token.isEmpty
       
        if hasToken && hasDomain {
            return PlaybackOptions(customDomain: item.customDomain, playbackToken: item.token)
        } else if hasToken {
            return PlaybackOptions(playbackToken: item.token)
        } else if hasDomain {
            return PlaybackOptions(customDomain: item.customDomain)
        } else {
            return PlaybackOptions()
        }
    }
   
    /// Builds the HLS stream URL for a playlist item directly,
    /// without relying on the SDK's internal AVPlayerItem extensions.
    private func buildPlaybackURL(for item: FastPixPlaylistItem) -> URL? {
        let host = (item.customDomain.isEmpty == false) ? item.customDomain : "stream.fastpix.com"
        var components = URLComponents()
        components.scheme = "https"
        components.host   = host
        components.path   = "/\(item.playbackId).m3u8"
       
        var queryItems: [URLQueryItem] = []
        if !item.token.isEmpty {
            queryItems.append(URLQueryItem(name: "token", value: item.token))
        }
        if !queryItems.isEmpty {
            components.queryItems = queryItems
        }
        return components.url
    }
}
